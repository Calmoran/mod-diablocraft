cmake_minimum_required(VERSION 3.16)
project(dc_tier_login_tests LANGUAGES CXX)
enable_testing()
# Compile the production helper with the installed core's actual comparison
# and same-phase early return, without linking a second worldserver.
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../dc_difficulty.cpp" MODULE_SOURCE)
string(REGEX MATCH "void ApplyPhase\\(WorldObject\\* object, uint32 phase, bool update\\)\n\\{[^}]*\\}" PHASE_HELPER "${MODULE_SOURCE}")
# The player's phase rule and the Sanctum's shared phase, cut out of production the same way.
string(REGEX MATCH "uint32 PlayerPhase\\(Player const\\* player\\)\n\\{[^}]*\\}" PLAYER_PHASE "${MODULE_SOURCE}")
string(REGEX MATCH "unsigned tier = player->GetGroup\\(\\) \\? [^;]*;" LOGIN_TIER "${MODULE_SOURCE}")
file(READ "${REPO}/mod-diablocraft/src/common/dc_world.h" WORLD_HEADER)
string(REGEX MATCH "constexpr unsigned SanctumPhase = [0-9]+;" SANCTUM_PHASE "${WORLD_HEADER}")
file(READ "${REPO}/azerothcore-wotlk/src/server/game/Entities/Unit/Unit.cpp" UNIT_SOURCE)
string(REGEX MATCH "if \\(newPhaseMask == GetPhaseMask\\(\\)\\)\n[ ]*return;" UNIT_EARLY_RETURN "${UNIT_SOURCE}")
file(READ "${REPO}/azerothcore-wotlk/src/server/game/Entities/Object/Object.h" OBJECT_SOURCE)
string(REGEX MATCH "bool InSamePhase\\(uint32 phasemask\\) const \\{[^}]*\\}" PHASE_COMPARISON "${OBJECT_SOURCE}")
if(NOT PLAYER_PHASE OR NOT LOGIN_TIER OR NOT SANCTUM_PHASE OR NOT PHASE_HELPER OR NOT UNIT_EARLY_RETURN OR NOT PHASE_COMPARISON)
    message(FATAL_ERROR "Core/module source changed; re-audit the phase regression harness")
endif()
configure_file(phase_login_tests.cpp.in phase_login_tests.cpp @ONLY)
add_executable(dc_tier_login_tests "${CMAKE_CURRENT_BINARY_DIR}/phase_login_tests.cpp")
target_compile_features(dc_tier_login_tests PRIVATE cxx_std_17)
add_test(NAME tier_login_phase_policy COMMAND dc_tier_login_tests)
